option(EF_BUILD_APP "Build the SDL3/OpenGL preview application" ON)

option(EF_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

target_include_directories(ef_core PUBLIC src)
target_link_libraries(ef_core PUBLIC nlohmann_json::nlohmann_json PNG::PNG ${EF_WEBP_TARGET})

function(ef_compile_options target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(EF_SANITIZERS)
            target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
ef_compile_options(ef_core)

add_library(ef_stb_vorbis STATIC src/audio/vorbis.c)
target_include_directories(ef_stb_vorbis PRIVATE "${EF_STB_VORBIS_INCLUDE_DIR}")
target_compile_definitions(ef_stb_vorbis PRIVATE EF_STB_VORBIS_HEADER="${EF_STB_VORBIS_HEADER}")
add_library(ef_audio src/audio/clip.cpp src/audio/mixer.cpp)
target_include_directories(ef_audio PUBLIC src PRIVATE "${EF_STB_VORBIS_INCLUDE_DIR}")
target_compile_definitions(ef_audio PRIVATE EF_STB_VORBIS_HEADER="${EF_STB_VORBIS_HEADER}")
target_link_libraries(ef_audio PRIVATE ef_stb_vorbis)
ef_compile_options(ef_audio)
target_link_libraries(ef_core PUBLIC ef_audio)
if(EF_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ef_stb_vorbis PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
